Student Marks Management System

"""
==============================================
  STUDENT MARKS MANAGEMENT SYSTEM
  Mini Project 1 - Python for Data Science
==============================================
"""

print("=" * 50)
print("   STUDENT MARKS MANAGEMENT SYSTEM")
print("=" * 50)

# Get number of students
num_students = int(input("\nHow many students? "))

# Store all student results
all_results = []

for student in range(1, num_students + 1):
    print(f"\n{'─' * 40}")
    print(f"  Enter details for Student {student}")
    print(f"{'─' * 40}")
    
    # Student Data Input
    name = input("Student Name: ")
    roll = input("Roll Number: ")
    
    # Marks Input (5 subjects)
    print("\nEnter marks for 5 subjects (out of 100 each):")
    subjects = ["Mathematics", "Science", "English", "Hindi", "Computer"]
    marks = []
    
    for subject in subjects:
        while True:
            mark = float(input(f"  {subject}: "))
            if 0 <= mark <= 100:
                marks.append(mark)
                break
            else:
                print("  Invalid! Enter marks between 0-100")
    
    # Marks Calculation
    total = sum(marks)
    max_marks = len(subjects) * 100
    percentage = (total / max_marks) * 100
    average = total / len(subjects)
    
    # Grade Assignment
    if percentage >= 90:
        grade = "A+"
        remark = "Outstanding"
    elif percentage >= 80:
        grade = "A"
        remark = "Excellent"
    elif percentage >= 70:
        grade = "B"
        remark = "Very Good"
    elif percentage >= 60:
        grade = "C"
        remark = "Good"
    elif percentage >= 50:
        grade = "D"
        remark = "Average"
    elif percentage >= 40:
        grade = "E"
        remark = "Below Average"
    else:
        grade = "F"
        remark = "FAIL"
    
    # Pass/Fail status
    status = "PASSED" if percentage >= 40 else "FAILED"
    
    # Find highest and lowest scoring subjects
    highest_idx = marks.index(max(marks))
    lowest_idx = marks.index(min(marks))
    
    # Result Display
    print(f"\n{'═' * 50}")
    print(f"         RESULT CARD - {name.upper()}")
    print(f"{'═' * 50}")
    print(f"  Roll Number  : {roll}")
    print(f"  Student Name : {name}")
    print(f"{'─' * 50}")
    print(f"  {'Subject':<15} {'Marks':<10} {'Status'}")
    print(f"{'─' * 50}")
    
    for i, subject in enumerate(subjects):
        sub_status = "Pass" if marks[i] >= 40 else "Fail"
        print(f"  {subject:<15} {marks[i]:<10.1f} {sub_status}")
    
    print(f"{'─' * 50}")
    print(f"  Total Marks  : {total:.1f}/{max_marks}")
    print(f"  Percentage   : {percentage:.2f}%")
    print(f"  Average      : {average:.2f}")
    print(f"  Grade        : {grade}")
    print(f"  Remark       : {remark}")
    print(f"  Status       : {status}")
    print(f"{'─' * 50}")
    print(f"  Best Subject : {subjects[highest_idx]} ({marks[highest_idx]})")
    print(f"  Weak Subject : {subjects[lowest_idx]} ({marks[lowest_idx]})")
    print(f"{'═' * 50}")
    
    # Store result for report
    all_results.append({
        "name": name,
        "roll": roll,
        "total": total,
        "percentage": percentage,
        "grade": grade,
        "status": status
    })

# Basic Report Generation
print(f"\n\n{'═' * 60}")
print(f"          OVERALL CLASS REPORT")
print(f"{'═' * 60}")
print(f"  {'Name':<15} {'Roll':<8} {'Total':<8} {'%age':<8} {'Grade':<6} {'Status'}")
print(f"{'─' * 60}")

passed_count = 0
for result in all_results:
    print(f"  {result['name']:<15} {result['roll']:<8} {result['total']:<8.0f} {result['percentage']:<8.1f} {result['grade']:<6} {result['status']}")
    if result['status'] == "PASSED":
        passed_count += 1

print(f"{'─' * 60}")
print(f"  Total Students : {num_students}")
print(f"  Passed         : {passed_count}")
print(f"  Failed         : {num_students - passed_count}")
print(f"  Pass Rate      : {(passed_count/num_students)*100:.1f}%")
print(f"{'═' * 60}")

Code Explanation

Step-by-Step Breakdown

1. Data Input Phase

  • Uses input() for student info
  • Loop with validation ensures marks are 0-100
  • Stores marks in a list for easy calculation

2. Processing Phase

  • sum() calculates total marks
  • Percentage = (total/max) × 100
  • if-elif-else chain assigns grades
  • max() and min() find best/weak subjects

3. Output Phase

  • Formatted output with f-strings
  • Table-like display using string formatting
  • Class report with pass/fail statistics
