Module 1: Face Capture

Student Face Data Collection

import cv2
import os
import csv
from datetime import datetime

class FaceCapture:
    def __init__(self):
        self.face_cascade = cv2.CascadeClassifier(
            cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
        )
        self.data_dir = "student_faces"
        os.makedirs(self.data_dir, exist_ok=True)
        
        # Student database
        self.db_file = "students.csv"
        if not os.path.exists(self.db_file):
            with open(self.db_file, 'w', newline='') as f:
                writer = csv.writer(f)
                writer.writerow(['ID', 'Name', 'Roll_No', 'Registration_Date'])
    
    def register_student(self, name, roll_no, num_images=50):
        """Capture face images for a new student"""
        student_id = f"STU_{roll_no}"
        student_dir = os.path.join(self.data_dir, student_id)
        os.makedirs(student_dir, exist_ok=True)
        
        # Save to database
        with open(self.db_file, 'a', newline='') as f:
            writer = csv.writer(f)
            writer.writerow([student_id, name, roll_no, 
                           datetime.now().strftime("%Y-%m-%d")])
        
        # Capture faces
        cap = cv2.VideoCapture(0)
        count = 0
        
        print(f"\nRegistering: {name} (Roll: {roll_no})")
        print(f"Capturing {num_images} face images...")
        print("Look at the camera. Press 'q' to stop early.")
        
        while count < num_images:
            ret, frame = cap.read()
            if not ret:
                break
            
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            faces = self.face_cascade.detectMultiScale(gray, 1.3, 5)
            
            for (x, y, w, h) in faces:
                count += 1
                face = gray[y:y+h, x:x+w]
                face = cv2.resize(face, (200, 200))
                cv2.imwrite(f"{student_dir}/{count}.jpg", face)
                
                # Visual feedback
                cv2.rectangle(frame, (x, y), (x+w, y+h), (0, 255, 0), 2)
                progress = f"Capturing: {count}/{num_images}"
                cv2.putText(frame, progress, (10, 30), 
                           cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
            
            cv2.imshow("Face Registration", frame)
            if cv2.waitKey(1) & 0xFF == ord('q'):
                break
        
        cap.release()
        cv2.destroyAllWindows()
        print(f"Registration complete! {count} images captured.")
        return student_id

Module 2: Model Training

Train Face Recognition Model

import cv2
import numpy as np
import os
import pickle

class FaceTrainer:
    def __init__(self):
        self.recognizer = cv2.face.LBPHFaceRecognizer_create()
        self.data_dir = "student_faces"
        self.model_file = "attendance_model.yml"
        self.labels_file = "label_map.pkl"
    
    def train(self):
        """Train the face recognition model"""
        faces = []
        labels = []
        label_map = {}
        label_id = 0
        
        print("Training face recognition model...")
        
        for student_id in sorted(os.listdir(self.data_dir)):
            student_path = os.path.join(self.data_dir, student_id)
            if not os.path.isdir(student_path):
                continue
            
            label_map[label_id] = student_id
            img_count = 0
            
            for img_name in os.listdir(student_path):
                img_path = os.path.join(student_path, img_name)
                img = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE)
                if img is not None:
                    img = cv2.resize(img, (200, 200))
                    faces.append(img)
                    labels.append(label_id)
                    img_count += 1
            
            print(f"  {student_id}: {img_count} images loaded")
            label_id += 1
        
        # Train LBPH recognizer
        self.recognizer.train(faces, np.array(labels))
        
        # Save model
        self.recognizer.save(self.model_file)
        with open(self.labels_file, 'wb') as f:
            pickle.dump(label_map, f)
        
        print(f"\nModel trained with {len(faces)} total images")
        print(f"Registered students: {len(label_map)}")
        print(f"Model saved: {self.model_file}")
        
        return label_map

Module 3: Face Recognition

Real-time Recognition & Attendance Marking

import cv2
import pickle
import csv
from datetime import datetime, date

class AttendanceSystem:
    def __init__(self):
        self.recognizer = cv2.face.LBPHFaceRecognizer_create()
        self.recognizer.read("attendance_model.yml")
        
        with open("label_map.pkl", 'rb') as f:
            self.label_map = pickle.load(f)
        
        self.face_cascade = cv2.CascadeClassifier(
            cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
        )
        
        self.today_attendance = {}
        self.attendance_file = f"attendance_{date.today().strftime('%Y_%m_%d')}.csv"
        self.confidence_threshold = 65
    
    def mark_attendance(self, student_id, name):
        """Mark attendance for recognized student"""
        if student_id not in self.today_attendance:
            time_now = datetime.now().strftime("%H:%M:%S")
            self.today_attendance[student_id] = {
                'name': name,
                'time': time_now,
                'date': date.today().strftime("%Y-%m-%d")
            }
            
            # Save to CSV
            with open(self.attendance_file, 'a', newline='') as f:
                writer = csv.writer(f)
                writer.writerow([student_id, name, 
                               date.today().strftime("%Y-%m-%d"), time_now])
            
            print(f"  ✓ Attendance marked: {name} at {time_now}")
            return True
        return False
    
    def run(self):
        """Start real-time face recognition attendance"""
        cap = cv2.VideoCapture(0)
        
        print("\n" + "="*50)
        print("  SMART ATTENDANCE SYSTEM - ACTIVE")
        print("="*50)
        print("Press 'q' to quit | 'r' to view report\n")
        
        while True:
            ret, frame = cap.read()
            if not ret:
                break
            
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            faces = self.face_cascade.detectMultiScale(gray, 1.3, 5)
            
            for (x, y, w, h) in faces:
                face_roi = gray[y:y+h, x:x+w]
                face_roi = cv2.resize(face_roi, (200, 200))
                
                label, confidence = self.recognizer.predict(face_roi)
                
                if confidence < self.confidence_threshold:
                    student_id = self.label_map[label]
                    # Load name from students.csv
                    name = student_id  # Simplified
                    
                    color = (0, 255, 0)
                    self.mark_attendance(student_id, name)
                    text = f"{name} ({100-confidence:.0f}%)"
                else:
                    color = (0, 0, 255)
                    text = "Unknown"
                
                cv2.rectangle(frame, (x, y), (x+w, y+h), color, 2)
                cv2.putText(frame, text, (x, y-10),
                           cv2.FONT_HERSHEY_SIMPLEX, 0.7, color, 2)
            
            # Show attendance count
            count_text = f"Present Today: {len(self.today_attendance)}"
            cv2.putText(frame, count_text, (10, 30),
                       cv2.FONT_HERSHEY_SIMPLEX, 0.8, (255, 255, 255), 2)
            
            cv2.imshow("Attendance System", frame)
            
            key = cv2.waitKey(1) & 0xFF
            if key == ord('q'):
                break
            elif key == ord('r'):
                self.print_report()
        
        cap.release()
        cv2.destroyAllWindows()
        self.print_report()
    
    def print_report(self):
        """Print today's attendance report"""
        print("\n" + "="*50)
        print(f"  ATTENDANCE REPORT - {date.today()}")
        print("="*50)
        if self.today_attendance:
            for sid, info in self.today_attendance.items():
                print(f"  {info['name']:<20} {info['time']}")
            print(f"\n  Total Present: {len(self.today_attendance)}")
        else:
            print("  No attendance recorded yet.")
        print("="*50)

Module 4: Attendance Management

Attendance Records & Data Storage

import pandas as pd
from datetime import date, timedelta

class AttendanceManager:
    def __init__(self):
        self.records_dir = "attendance_records"
        os.makedirs(self.records_dir, exist_ok=True)
    
    def get_attendance_report(self, start_date=None, end_date=None):
        """Generate attendance report for date range"""
        all_records = []
        
        for file in os.listdir(self.records_dir):
            if file.endswith('.csv'):
                df = pd.read_csv(os.path.join(self.records_dir, file))
                all_records.append(df)
        
        if all_records:
            report = pd.concat(all_records, ignore_index=True)
            report.columns = ['Student_ID', 'Name', 'Date', 'Time']
            
            if start_date and end_date:
                report = report[(report['Date'] >= start_date) & 
                               (report['Date'] <= end_date)]
            
            return report
        return pd.DataFrame()
    
    def get_student_attendance(self, student_id):
        """Get attendance record for specific student"""
        report = self.get_attendance_report()
        if not report.empty:
            return report[report['Student_ID'] == student_id]
        return pd.DataFrame()
    
    def calculate_percentage(self, student_id, total_days):
        """Calculate attendance percentage"""
        student_records = self.get_student_attendance(student_id)
        present_days = len(student_records['Date'].unique())
        percentage = (present_days / total_days) * 100
        return percentage

Module 5: Data Visualization Dashboard

Attendance Analytics

import matplotlib.pyplot as plt
import pandas as pd
import numpy as np

def create_attendance_dashboard(attendance_data):
    """Generate visual attendance dashboard"""
    fig, axes = plt.subplots(2, 2, figsize=(14, 10))
    fig.suptitle("Smart Attendance System - Dashboard", 
                fontsize=16, fontweight='bold')
    
    # 1. Daily attendance trend
    daily = attendance_data.groupby('Date').size()
    axes[0, 0].plot(daily.index, daily.values, 'o-', color='#ff1b6b')
    axes[0, 0].set_title("Daily Attendance Trend")
    axes[0, 0].set_xlabel("Date")
    axes[0, 0].set_ylabel("Students Present")
    axes[0, 0].tick_params(axis='x', rotation=45)
    
    # 2. Student-wise attendance
    student_count = attendance_data.groupby('Name')['Date'].nunique()
    axes[0, 1].barh(student_count.index, student_count.values, color='#667eea')
    axes[0, 1].set_title("Student-wise Attendance")
    axes[0, 1].set_xlabel("Days Present")
    
    # 3. Time distribution
    attendance_data['Hour'] = pd.to_datetime(attendance_data['Time']).dt.hour
    axes[1, 0].hist(attendance_data['Hour'], bins=12, color='#f093fb', edgecolor='black')
    axes[1, 0].set_title("Check-in Time Distribution")
    axes[1, 0].set_xlabel("Hour of Day")
    axes[1, 0].set_ylabel("Frequency")
    
    # 4. Attendance percentage pie
    total_students = attendance_data['Name'].nunique()
    total_days = attendance_data['Date'].nunique()
    avg_attendance = daily.mean()
    absent_avg = total_students - avg_attendance
    
    axes[1, 1].pie([avg_attendance, absent_avg], 
                   labels=['Present', 'Absent'],
                   colors=['#4caf50', '#f44336'],
                   autopct='%1.1f%%', startangle=90)
    axes[1, 1].set_title("Average Attendance Rate")
    
    plt.tight_layout()
    plt.savefig("attendance_dashboard.png", dpi=150, bbox_inches='tight')
    plt.show()
    print("Dashboard saved as 'attendance_dashboard.png'")

Q1: What are the main modules of this attendance system?

Show Answer

Q2: Why use LBPH for face recognition?

Show Answer

LBPH (Local Binary Patterns Histograms) works well with small datasets, handles lighting variations, doesn't require massive computing power, and is included in OpenCV.

Q3: How does the system prevent duplicate attendance?

Show Answer

By checking if student_id already exists in today_attendance dictionary before marking. Each student can only be marked once per day.

Q4: What is the confidence threshold and why is it important?

Show Answer

It's the maximum acceptable distance between predicted and actual face. Lower = stricter matching. Prevents wrong identifications.

Q5: How would you improve this system for production?

Show Answer

Add anti-spoofing, use deep learning (FaceNet), database storage, web interface, notification system, backup, security measures.

Q6: What ML concepts from this course are used in this project?

Data collection, preprocessing, model training (LBPH), prediction, evaluation (confidence scores), model saving (pickle/yml), and data visualization.
