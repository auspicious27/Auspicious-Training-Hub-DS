Complete Project Code

Step 1: Create & Load Dataset

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Create sample student dataset
np.random.seed(42)
n_students = 50

data = {
    "Student_ID": [f"STU{str(i).zfill(3)}" for i in range(1, n_students+1)],
    "Name": [f"Student_{i}" for i in range(1, n_students+1)],
    "Math": np.random.randint(35, 100, n_students),
    "Science": np.random.randint(30, 100, n_students),
    "English": np.random.randint(40, 100, n_students),
    "Hindi": np.random.randint(35, 100, n_students),
    "Computer": np.random.randint(45, 100, n_students),
    "Attendance": np.random.randint(60, 100, n_students)
}

df = pd.DataFrame(data)

# Add some missing values for realism
df.loc[5, "Math"] = np.nan
df.loc[12, "Science"] = np.nan
df.loc[20, "English"] = np.nan

print("Dataset Shape:", df.shape)
print("\nFirst 5 rows:")
print(df.head())
print("\nDataset Info:")
print(df.info())

Step 2: Data Cleaning

# Check missing values
print("Missing Values:")
print(df.isnull().sum())

# Fill missing values with column mean
df["Math"].fillna(df["Math"].mean(), inplace=True)
df["Science"].fillna(df["Science"].mean(), inplace=True)
df["English"].fillna(df["English"].mean(), inplace=True)

print("\nAfter cleaning - Missing Values:", df.isnull().sum().sum())

Step 3: Data Analysis

# Calculate total and percentage
subjects = ["Math", "Science", "English", "Hindi", "Computer"]
df["Total"] = df[subjects].sum(axis=1)
df["Percentage"] = (df["Total"] / 500) * 100

# Assign grades
def assign_grade(pct):
    if pct >= 90: return "A+"
    elif pct >= 80: return "A"
    elif pct >= 70: return "B"
    elif pct >= 60: return "C"
    elif pct >= 40: return "D"
    else: return "F"

df["Grade"] = df["Percentage"].apply(assign_grade)
df["Status"] = df["Percentage"].apply(lambda x: "Pass" if x >= 40 else "Fail")

# Statistical Summary
print("\n" + "="*50)
print("       PERFORMANCE ANALYSIS REPORT")
print("="*50)
print(f"\nTotal Students: {len(df)}")
print(f"Passed: {len(df[df['Status']=='Pass'])}")
print(f"Failed: {len(df[df['Status']=='Fail'])}")
print(f"\nClass Average: {df['Percentage'].mean():.2f}%")
print(f"Highest Score: {df['Percentage'].max():.2f}%")
print(f"Lowest Score: {df['Percentage'].min():.2f}%")

# Subject-wise analysis
print("\nSubject-wise Average:")
for subject in subjects:
    print(f"  {subject}: {df[subject].mean():.2f}")

# Grade distribution
print("\nGrade Distribution:")
print(df["Grade"].value_counts())

Step 4: Identify Top Performers

# Top 5 students
top_5 = df.nlargest(5, "Percentage")[["Name", "Total", "Percentage", "Grade"]]
print("\nTop 5 Students:")
print(top_5.to_string(index=False))

# Students who need improvement (below 50%)
weak = df[df["Percentage"] < 50][["Name", "Percentage", "Grade"]]
print(f"\nStudents needing improvement: {len(weak)}")
print(weak.to_string(index=False))

Data Visualization

Step 5: Creating Charts

import matplotlib.pyplot as plt

# Create a figure with multiple subplots
fig, axes = plt.subplots(2, 2, figsize=(14, 10))
fig.suptitle("Student Performance Dashboard", fontsize=16, fontweight='bold')

# 1. Grade Distribution (Bar Chart)
grade_counts = df["Grade"].value_counts().sort_index()
axes[0, 0].bar(grade_counts.index, grade_counts.values, color=['#ff1b6b', '#ff6b6b', '#ffa726', '#66bb6a', '#42a5f5', '#ab47bc'])
axes[0, 0].set_title("Grade Distribution")
axes[0, 0].set_xlabel("Grade")
axes[0, 0].set_ylabel("Number of Students")

# 2. Subject-wise Average (Bar Chart)
subject_avg = df[subjects].mean()
axes[0, 1].barh(subjects, subject_avg, color='#667eea')
axes[0, 1].set_title("Subject-wise Average Marks")
axes[0, 1].set_xlabel("Average Marks")

# 3. Percentage Distribution (Histogram)
axes[1, 0].hist(df["Percentage"], bins=10, color='#f093fb', edgecolor='black')
axes[1, 0].set_title("Percentage Distribution")
axes[1, 0].set_xlabel("Percentage")
axes[1, 0].set_ylabel("Frequency")
axes[1, 0].axvline(df["Percentage"].mean(), color='red', linestyle='--', label=f'Mean: {df["Percentage"].mean():.1f}%')
axes[1, 0].legend()

# 4. Attendance vs Percentage (Scatter Plot)
axes[1, 1].scatter(df["Attendance"], df["Percentage"], c='#ff1b6b', alpha=0.6)
axes[1, 1].set_title("Attendance vs Performance")
axes[1, 1].set_xlabel("Attendance (%)")
axes[1, 1].set_ylabel("Percentage")

plt.tight_layout()
plt.savefig("performance_dashboard.png", dpi=150, bbox_inches='tight')
plt.show()

print("\nDashboard saved as 'performance_dashboard.png'")

Step 6: Generate Final Report

# Save results to CSV
df.to_csv("student_results.csv", index=False)

# Print final summary
print("\n" + "="*50)
print("    FINAL ANALYSIS SUMMARY")
print("="*50)
print(f"  Total Students Analyzed: {len(df)}")
print(f"  Pass Rate: {(len(df[df['Status']=='Pass'])/len(df))*100:.1f}%")
print(f"  Class Average: {df['Percentage'].mean():.2f}%")
print(f"  Best Subject: {subject_avg.idxmax()} ({subject_avg.max():.1f})")
print(f"  Weakest Subject: {subject_avg.idxmin()} ({subject_avg.min():.1f})")
print(f"  Topper: {df.loc[df['Percentage'].idxmax(), 'Name']} ({df['Percentage'].max():.1f}%)")
print(f"\n  Results saved to 'student_results.csv'")
print("="*50)

Q1: What does df.nlargest(5, "Marks") do?

Show Answer

Q2: How do you calculate column-wise sum in Pandas?

Show Answer

df[columns].sum(axis=1) for row-wise sum, df[columns].sum(axis=0) for column-wise sum.

Q3: What does the .apply() method do?

Show Answer

Applies a function to each element/row of a Series or DataFrame. Example: df["col"].apply(func)

Show Answer
