Face Detection using OpenCV

Haar Cascade Face Detection

import cv2

# Load pre-trained face detector
face_cascade = cv2.CascadeClassifier(
    cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
)

# Read image
img = cv2.imread("group_photo.jpg")
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# Detect faces
faces = face_cascade.detectMultiScale(
    gray,
    scaleFactor=1.1,    # How much image size is reduced
    minNeighbors=5,     # How many neighbors for valid detection
    minSize=(30, 30)    # Minimum face size
)

print(f"Found {len(faces)} faces!")

# Draw rectangles around faces
for (x, y, w, h) in faces:
    cv2.rectangle(img, (x, y), (x+w, y+h), (0, 255, 0), 2)
    cv2.putText(img, "Face", (x, y-10), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)

Capture Face Images from Webcam

Building Face Data Collector

import cv2
import os

# Create directory for face data
def capture_faces(name, num_images=50):
    path = f"face_data/{name}"
    os.makedirs(path, exist_ok=True)
    
    face_cascade = cv2.CascadeClassifier(
        cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
    )
    
    cap = cv2.VideoCapture(0)  # Open webcam
    count = 0
    
    print(f"Capturing face data for: {name}")
    print("Look at the camera. Press 'q' to quit.")
    
    while count < num_images:
        ret, frame = cap.read()
        if not ret:
            break
            
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        faces = face_cascade.detectMultiScale(gray, 1.3, 5)
        
        for (x, y, w, h) in faces:
            count += 1
            face_img = gray[y:y+h, x:x+w]
            face_img = cv2.resize(face_img, (200, 200))
            
            # Save face image
            cv2.imwrite(f"{path}/{count}.jpg", face_img)
            
            # Draw rectangle
            cv2.rectangle(frame, (x, y), (x+w, y+h), (0, 255, 0), 2)
            cv2.putText(frame, f"Captured: {count}/{num_images}", 
                       (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        
        cv2.imshow("Capturing Faces", frame)
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
    
    cap.release()
    cv2.destroyAllWindows()
    print(f"Captured {count} images for {name}")

Face Recognition Model

Training Face Recognition

import cv2
import numpy as np
import os

def train_face_recognizer():
    face_data_path = "face_data"
    faces = []
    labels = []
    label_map = {}
    label_id = 0
    
    # Load all face images
    for person_name in os.listdir(face_data_path):
        person_path = os.path.join(face_data_path, person_name)
        if not os.path.isdir(person_path):
            continue
            
        label_map[label_id] = person_name
        
        for img_name in os.listdir(person_path):
            img_path = os.path.join(person_path, img_name)
            img = cv2.imread(img_path, cv2.IMREAD_GRAYSCALE)
            if img is not None:
                faces.append(img)
                labels.append(label_id)
        
        label_id += 1
    
    # Train LBPH Face Recognizer
    recognizer = cv2.face.LBPHFaceRecognizer_create()
    recognizer.train(faces, np.array(labels))
    
    # Save trained model
    recognizer.save("face_model.yml")
    np.save("label_map.npy", label_map)
    
    print(f"Model trained with {len(faces)} images")
    print(f"Registered users: {list(label_map.values())}")
    
    return recognizer, label_map

Real-time Face Recognition

Live Face Identification

import cv2
import numpy as np
from datetime import datetime

def recognize_faces():
    # Load trained model
    recognizer = cv2.face.LBPHFaceRecognizer_create()
    recognizer.read("face_model.yml")
    label_map = np.load("label_map.npy", allow_pickle=True).item()
    
    face_cascade = cv2.CascadeClassifier(
        cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
    )
    
    cap = cv2.VideoCapture(0)
    attendance = {}
    
    print("Face Recognition Active. Press 'q' to quit.")
    
    while True:
        ret, frame = cap.read()
        if not ret:
            break
        
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        faces = face_cascade.detectMultiScale(gray, 1.3, 5)
        
        for (x, y, w, h) in faces:
            face_roi = gray[y:y+h, x:x+w]
            face_roi = cv2.resize(face_roi, (200, 200))
            
            # Predict
            label, confidence = recognizer.predict(face_roi)
            
            if confidence < 70:  # Confidence threshold
                name = label_map[label]
                color = (0, 255, 0)
                text = f"{name} ({confidence:.0f}%)"
                
                # Mark attendance
                if name not in attendance:
                    attendance[name] = datetime.now().strftime("%H:%M:%S")
                    print(f"Attendance marked: {name} at {attendance[name]}")
            else:
                name = "Unknown"
                color = (0, 0, 255)
                text = "Unknown"
            
            cv2.rectangle(frame, (x, y), (x+w, y+h), color, 2)
            cv2.putText(frame, text, (x, y-10), 
                       cv2.FONT_HERSHEY_SIMPLEX, 0.7, color, 2)
        
        cv2.imshow("Face Recognition", frame)
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
    
    cap.release()
    cv2.destroyAllWindows()
    
    # Print attendance
    print("\n--- Attendance Record ---")
    for name, time in attendance.items():
        print(f"  {name}: {time}")

Day 13 Assessment

Q1: What is OpenCV used for?

Show Answer

Computer Vision tasks: image processing, face detection/recognition, object detection, video analysis.

Q2: What is Haar Cascade?

Show Answer

A pre-trained face detection model that uses cascading classifiers to detect faces in images. OpenCV provides it built-in.

Q3: What does cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) do?

Show Answer

Converts a color (BGR) image to grayscale. Face detection works better on grayscale images.

Q4: What is LBPH?

Show Answer

Local Binary Patterns Histograms — a face recognition algorithm that analyzes texture patterns in face images to identify people.

Q5: Why do we capture multiple images per person?

Show Answer

To capture different angles, expressions, and lighting conditions. More training data = better recognition accuracy.

Q6: What does confidence value in face recognition mean?

Show Answer

Lower confidence = better match. A confidence of 0 means perfect match. Below 70 is typically considered a valid recognition.

Q7: How to open webcam in OpenCV?

Show Solution
cap = cv2.VideoCapture(0)  # 0 = default webcam
ret, frame = cap.read()    # Read a frame
cap.release()              # Release webcam

Q8: What does detectMultiScale() return?

Show Answer

A list of rectangles (x, y, width, height) for each detected face in the image.

Q9: How can this project be extended?

Show Answer

Add attendance logging to CSV, email notifications, multiple camera support, anti-spoofing, database integration, web interface.

Q10: Difference between face detection and face recognition?

Show Answer

Detection: Finding WHERE a face is in an image. Recognition: Identifying WHO the face belongs to.
